(function () {

    "use strict";

    function initializeFloatingChat() {

        const chatMessage = document.getElementById("info-chat-interval");
        const chatButton = document.getElementById("chatButton");

        if (!chatMessage || !chatButton) {
            return;
        }

        /* =================================================
           CONFIGURACIÓN
        ================================================= */

        const initialVisibleTime = 6000;
        const hoverVisibleTime = 2500;

        let hideTimer = null;

        function showMessage() {
            window.clearTimeout(hideTimer);

            chatMessage.classList.remove("chat-hidden");
            chatMessage.setAttribute("aria-hidden", "false");
        }

        function hideMessage() {
            chatMessage.classList.add("chat-hidden");
            chatMessage.setAttribute("aria-hidden", "true");
        }

        hideTimer = window.setTimeout(
            hideMessage,
            initialVisibleTime
        );

        chatButton.addEventListener("mouseenter", showMessage);

        chatButton.addEventListener("mouseleave", function () {
            window.clearTimeout(hideTimer);

            hideTimer = window.setTimeout(
                hideMessage,
                hoverVisibleTime
            );
        });

        chatButton.addEventListener("focus", showMessage);

        chatButton.addEventListener("blur", function () {
            window.clearTimeout(hideTimer);

            hideTimer = window.setTimeout(
                hideMessage,
                hoverVisibleTime
            );
        });
    }

    if (document.readyState === "loading") {
        document.addEventListener(
            "DOMContentLoaded",
            initializeFloatingChat
        );
    } else {
        initializeFloatingChat();
    }

})();
</script>

 

<script>
    const checkAuthCountry = (privateAreaUrl, acceptTermsUrl) => {
        const currentCountry = "CO";
        const maxRetries = 1;
        const retryCount = Number(sessionStorage.getItem('retryCount') || 0);

        console.log("[checkAuthCountry] init", {
            currentCountry,
            privateAreaUrl,
            acceptTermsUrl,
            retryCount
        });


        if (retryCount >= maxRetries) {
            console.error("[checkAuthCountry] Máximo número de recargas alcanzado. Abortando.");
            return;
        }

        jQuery.ajaxSetup({ async: false });

        commonLatamAjaxCall(null, '', 'getProfileUID', 'POST', function (response, xhr) {
            console.log("[checkAuthCountry] getProfileUID RAW:", {
                status: xhr.status,
                readyState: xhr.readyState
            });

            if (xhr.status !== 200) {
                console.error("[checkAuthCountry] getProfileUID no responde correctamente. Recargando página...");
                sessionStorage.setItem('retryCount', retryCount + 1);
                location.reload(true);
                return;
            }

            if (!response || response?.result === "KO") {
                console.log("[checkAuthCountry] Usuario no logueado o respuesta KO.");
                return;
            }

            let result = (typeof response === 'string') ? JSON.parse(response) : response;
            console.log("[checkAuthCountry] getProfileUID parsed:", result);

            const logoutButtonData = {
                "customerType": "personas",
                "username": result?.email || "",
                "socialOneHubMail": result?.email || "",
                "socialOneHubUid": result?.enelId || "",
                "socialOneHubName": result?.firstName || "",
                "socialOneHubLastName": result?.lastName || "",
                "socialOneHubMobile": result?.phoneNumber || ""
            };
            sessionStorage.setItem("logoutInfo", JSON.stringify(logoutButtonData));

            const userAttributes = result?.attributes || [];
            const hasAcceptedForCurrentCountry = userAttributes.some(att => att.country === currentCountry);

            console.log("[checkAuthCountry] attributes check:", {
                currentCountry,
                hasAcceptedForCurrentCountry
            });

            if (hasAcceptedForCurrentCountry) {
                console.log("[checkAuthCountry] REDIRECT → privateAreaUrl");
                window.location.href = privateAreaUrl;
            } else {
                console.log("[checkAuthCountry] REDIRECT → acceptTermsUrl");
                sessionStorage.setItem("enelID", result?.enelId || "");
                window.location.href = acceptTermsUrl;
            }
        });

        jQuery.ajaxSetup({ async: true });
    }
</script>

	const checkUserIsLogged = () =>{

			if (!isAuthenticated) {
				if (typeof onComplete === "function") onComplete(false);
				return;

					try {
						profile = JSON.parse(rawProfile);
					} catch (error) {
						profile = null;

  function getRedirectElement() {
    const all = document.querySelectorAll(".redirectUrlLogin");
    for (const el of all) {
      if (el?.dataset && (el.dataset.termsurl || el.dataset.completeregurl || el.dataset.redirecturl)) {
        return el;
      }
    }
    return all[0] || null;
  }

  function getRedirects(loginType) {
    const el = getRedirectElement();

    let complete = ds.completeregurl || "";
    if (loginType === "empresas" && complete) {
      complete = complete.replace("personas","empresas"); // corrección efectiva
    }
    return {
      terms: ds.termsurl || "",
      complete,
      home: ds.redirecturl || ""
    };
  }
