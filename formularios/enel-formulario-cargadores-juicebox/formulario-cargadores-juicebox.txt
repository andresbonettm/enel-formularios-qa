          s.linkTrackVars   = "eVar1,eVar2,eVar3,eVar45,prop1,events";
          s.linkTrackEvents = eventCode;

          s.eVar2  = actionName;

          s.eVar45 = ecidCampania;
          s.prop1  = actionName;

          console.error("Error Adobe Analytics:", e);
        }
      }
